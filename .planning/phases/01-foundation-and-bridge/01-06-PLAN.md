  <name>Task 1: Create build-and-deploy script</name>
  <files>frontend/scripts/build-and-deploy.sh</files>
  <action>
  Create scripts directory and deployment script:

  ```bash
  mkdir -p frontend/scripts
  ```

  Create build-and-deploy.sh with proper error handling:

  ```bash
  #!/bin/bash
  set -e  # Exit on error

  # Colors for output
  GREEN='\033[0;32m'
  YELLOW='\033[1;33m'
  RED='\033[0;31m'
  NC='\033[0m' # No Color

  # Configuration
  PI_HOST="192.168.0.95"
  PI_USER="remus"
  PI_SSH_KEY="$HOME/.ssh/id_ed25519_pi"
  PI_APP_PATH="/home/remus/apps/platform/frontend"
  LOCAL_FRONTEND_PATH="$(cd "$(dirname "$0")/.." && pwd)"

  echo -e "${YELLOW}=== Platform Frontend Build & Deploy ===${NC}"

  # Step 1: Clean and build
  echo -e "${GREEN}[1/5] Cleaning Angular cache...${NC}"
  cd "$LOCAL_FRONTEND_PATH"
  rm -rf .angular/cache

  echo -e "${GREEN}[2/5] Building production bundle...${NC}"
  npx ng build --configuration production

  # Step 2: Validate build output
  echo -e "${GREEN}[3/5] Validating build output...${NC}"
  if [ ! -d "dist/browser" ]; then
    echo -e "${RED}ERROR: dist/browser not found. Build may have failed.${NC}"
    exit 1
  fi

  if [ ! -f "dist/browser/index.html" ]; then
    echo -e "${RED}ERROR: dist/browser/index.html not found. Build incomplete.${NC}"
    exit 1
  fi

  echo -e "${YELLOW}Build size:${NC}"
  du -sh dist/browser

  # Step 3: Stop PM2 service on Pi
  echo -e "${GREEN}[4/5] Stopping platform-frontend on Pi...${NC}"
  ssh -i "$PI_SSH_KEY" "$PI_USER@$PI_HOST" "pm2 stop platform-frontend || true"

  # Step 4: Deploy to Pi
  echo -e "${GREEN}[5/5] Deploying to Pi...${NC}"

  # Backup existing dist (optional safety measure)
  ssh -i "$PI_SSH_KEY" "$PI_USER@$PI_HOST" "cd $PI_APP_PATH && rm -rf dist.backup && mv dist dist.backup || true"

  # Copy new dist
  scp -i "$PI_SSH_KEY" -r dist/browser "$PI_USER@$PI_HOST:$PI_APP_PATH/dist"

  # Restart PM2 service
  echo -e "${GREEN}Restarting platform-frontend...${NC}"
  ssh -i "$PI_SSH_KEY" "$PI_USER@$PI_HOST" "pm2 restart platform-frontend"

  # Check status
  echo -e "${GREEN}Checking service status...${NC}"
  ssh -i "$PI_SSH_KEY" "$PI_USER@$PI_HOST" "pm2 status platform-frontend"

  echo -e "${GREEN}=== Deploy complete! ===${NC}"
  echo -e "${YELLOW}Frontend available at: https://thisisvillegas.com${NC}"
  ```

  Make script executable:
  ```bash
  chmod +x frontend/scripts/build-and-deploy.sh
  ```
