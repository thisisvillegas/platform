  <action>
  Create a standalone TypeScript script to clean up expired guests:

  ```typescript
  #!/usr/bin/env node
  /**
   * Guest Cleanup Script
   *
   * Deletes expired guest accounts from the database.
   * Designed to run nightly via cron.
   */

  import mongoose from 'mongoose';
  import dotenv from 'dotenv';
  import Guest from '../src/models/Guest';

  // Load environment variables
  dotenv.config();

  async function cleanupExpiredGuests() {
    try {
      console.log('[Guest Cleanup] Starting cleanup...');
      console.log('[Guest Cleanup] Time:', new Date().toISOString());

      // Connect to MongoDB
      const mongoUri = process.env.MONGODB_URI;
      if (!mongoUri) {
        throw new Error('MONGODB_URI not configured');
      }

      await mongoose.connect(mongoUri);
      console.log('[Guest Cleanup] Connected to MongoDB');

      // Find and delete expired guests
      const now = new Date();
      const result = await Guest.deleteMany({
        expiresAt: { $lt: now }
      });

      console.log(`[Guest Cleanup] Deleted ${result.deletedCount} expired guest account(s)`);

      // Log some stats
      const remainingCount = await Guest.countDocuments();
      console.log(`[Guest Cleanup] ${remainingCount} guest account(s) remaining`);

      // Disconnect
      await mongoose.disconnect();
      console.log('[Guest Cleanup] Cleanup complete');

      process.exit(0);

    } catch (error) {
      console.error('[Guest Cleanup] Error:', error);

      // Ensure we disconnect even on error
      try {
        await mongoose.disconnect();
      } catch (disconnectError) {
        console.error('[Guest Cleanup] Error disconnecting:', disconnectError);
      }

      process.exit(1);
    }
  }

  // Run cleanup
  cleanupExpiredGuests();
  ```
